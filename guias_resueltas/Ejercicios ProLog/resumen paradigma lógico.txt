%miembro\3: verifica si el elemento pasado está en la lista pasada
miembro(X,[X|_]):- !.
miembro(X,[_|Resto]):-
	miembro(X,Resto).

---------------------------------------------------------------------

%miembro_back\3: verifica si el elemento pasado está en la lista pasada
%si se le pasa una variable, devuelve por back_tracking todos los elementos

miembro(X,[X|_]).
miembro(X,[_|Resto]):-
	miembro(X,Resto).

%agregar\3:agrega un elemento a una lista


agregar(X,[],[X]):-!.

agregar(X,[X|Res],[X|Res]):-!.

agregar(X,[Y|L],[Y|Res]):-
	agregar(X,L,Res),
	!.

---------------------------------------------------------------------
%agregar_si\3: agrega al final de la lista cuando no sea miembro

agregar_si(X,[],[X]):- !.

agregar_si(X,[X|L],[X|L]):-
	!.

agregar_si(X,[Y|L],[Y|Res]):-
    	agregar_si(X,L,Res).

%insertar_ord\3: inserta en una lista ordenada

insertar_ord(X,[],[X]).
insertar_ord(X,[X2|L],[X2|R]):-
	X > X2,
	insertar_ord(X,L,R).

insertar_ord(X,[X2|L],[X,X2|L]):-
	X =< X2.

---------------------------------------------------------------------

%ordenar\2: ordena lista de numeros

insertar(X,[],[X]).
insertar(X,[X2|L],[X2|R]):-
	X > X2,
	insertar(X,L,R).

insertar(X,[X2|L],[X,X2|L]):-
	X =< X2.

ordenar([],[]).
ordenar([X|L1],ListaOrdenada):-
	ordenar(L1,L2),
	insertar(X,L2,ListaOrdenada).

%count_elem\3: cuente la cantidad de veces que aparece un elemento en una lista.

countelem(_,[],0):- !.
countelem(X,[Y|L],Cant):-
	X=Y,
	countelem(X,L,R2),
	Cant is R2+1,
	!.

countelem(X,[Y|L],Cant):-
	X\=Y,
	countelem(X,L,Cant),
	!.

%eliminar_dup\2: elimina los duplicados de una lista

eliminar(_,[],[]):-!.
eliminar(X,[Y|L],Lres):-
	X=Y,
	eliminar(X,L,Lres),
	!.
eliminar(X,[Y|L],[Y|Lres]):-
	X\=Y,
	eliminar(X,L,Lres),
	!.

eliminar_dup([],[]):-!.
eliminar_dup([X|L],[X|SinDup]):-
	eliminar(X,L,L2),
	eliminar_dup(L2,SinDup),
	!.

%suma_lista\3: suma elemento a elemento de ambas listas.

size([],0):-!.
size([A|B],R):-
	size(B,R2),
	R is R2 + 1,
	!.


suma_lista([],[],[]).
suma_lista(L1,L2,Res):-
	size(L1, R1),
	size(L2,R2),
	R1\=R2,
	!,
	fail.

suma_lista([X|L1],[Y|L2],[R|Res]):-
	size(L1,R1),
	size(L2,R2),
	R1=R2,
	suma_lista(L1,L2,Res),
	R is X+Y,
	!.

%union\3: devuelve la union de dos conjuntos

eliminar(_,[],[]):-!.
eliminar(X,[X|L],Lres):-
	eliminar(X,L,Lres),
	!.
eliminar(X,[Y|L],[Y|Lres]):-
	eliminar(X,L,Lres),
	!.

eliminar_dup([],[]):-!.
eliminar_dup([X|L],[X|SinDup]):-
	eliminar(X,L,L2),
	eliminar_dup(L2,SinDup),
	!.

concatenar([],L2,L2).
concatenar([X|L1],L2,[X|L3]):-
	concatenar(L1,L2,L3).
union(L1,L2,Res):-
	concatenar(L1,L2,Res1),
	eliminar_dup(Res1,Res).

%interseccion\3: devuelve la interseccion de dos conjuntos

miembro(X,[X|_]):- !.
miembro(X,[_|Resto]):-
	miembro(X,Resto).


interseccion([],[],[]):- !.
interseccion([_],[],[]):-
	!.
interseccion([],L2,[]):- !.

interseccion([X|L1],L2,[X|Res]):-
	miembro(X,L2),
	interseccion(L1,L2,Res),
	!.

interseccion([X|L1],L2,Res):-
	interseccion(L1,L2,Res).

%diferencia\3: devuelve la diferencia entre dos conjuntos

eliminar(_,[],[]):-!.
eliminar(X,[X|L],Lres):-
	eliminar(X,L,Lres),
	!.
eliminar(X,[Y|L],[Y|Lres]):-
	eliminar(X,L,Lres),
	!.


diferencia(L1,[],L1):- !.
diferencia([],_,[]):- !.

diferencia(L1,[Y|L2],Dif):-
	eliminar(Y,L1,Dif2),
	diferencia(Dif2,L2,Dif).

caminos([X], [X]):-!.
caminos([X|L], [X|L1]) :- miembro(C, L), caminos(C, L1).
